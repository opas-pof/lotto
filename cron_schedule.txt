# Cron Schedule สำหรับรัน scraper อัตโนมัติ
# ใช้กับ Linux/Mac หรือ cloud services ที่รองรับ cron

# รันทุกวันจันทร์, พุธ, ศุกร์ เวลา 20:30 น. (ตามเวลาประเทศไทย)
# หมายเหตุ: ต้องปรับ timezone ตาม server
30 20 * * 1,3,5 cd /path/to/lotto && python scheduler.py >> scraper.log 2>&1

# หรือใช้เวลา UTC (Thailand = UTC+7)
# รันทุกวันจันทร์, พุธ, ศุกร์ เวลา 13:30 UTC (= 20:30 น. ไทย)
30 13 * * 1,3,5 cd /path/to/lotto && python scheduler.py >> scraper.log 2>&1

# รันทุกวันเวลา 20:30 น. (ถ้าต้องการ)
30 20 * * * cd /path/to/lotto && python scheduler.py >> scraper.log 2>&1
